cmake_minimum_required(VERSION 3.20)
project(CoreSQLConsumer LANGUAGES CXX)
set(CORESQL_TEST_VERSION 0.1.0 CACHE STRING "Requested package version")
set(CORESQL_TEST_COMPONENTS core sql vector timestamp spatial json graph date decimal CACHE STRING "Requested components")
find_package(CoreSQL ${CORESQL_TEST_VERSION} EXACT REQUIRED COMPONENTS ${CORESQL_TEST_COMPONENTS})
add_executable(package_core core.cpp)
target_link_libraries(package_core PRIVATE CoreSQL::core)
add_executable(package_addons addons.cpp)
target_link_libraries(package_addons PRIVATE CoreSQL::sql CoreSQL::vector CoreSQL::timestamp CoreSQL::spatial CoreSQL::json CoreSQL::graph)
enable_testing()
add_test(NAME package_core COMMAND package_core)
add_test(NAME package_addons COMMAND package_addons)
